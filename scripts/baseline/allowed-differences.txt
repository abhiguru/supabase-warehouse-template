# Expected differences between the chain-built database (A) and the baseline-built database (B).
# Format: "<A|B>[@supabase|@pg17] <regex>" (see catalog-diff.mjs). A = the line exists only in A, B = only in B.
# @supabase / @pg17 limit an entry to a run whose B database is on that image. An entry that applies to a run and matches
# nothing FAILS the run. Every entry has a comment that says why the difference is expected (design item18-design.md
# sections 13 and 14). A regex cannot end in a space (it is trimmed): end it with a class such as [A-Z].
# The owner change of auth.email(), auth.role() and auth.uid() is NOT listed: catalog-diff.mjs rewrites side A for those
# three functions only (normaliseOwnerFacts), so a changed body or a lost grant still shows.

# Roles the new bootstrap does not create (design 13, 14d: "no postgres entries"; M3: no dashboard_user grants).
# The chain-built database still carries the image's grants to them; the baseline carries none.
A ^acl\|[a-z -]+\|[^|]+\|[a-z_]+ -> (postgres|dashboard_user|supabase_auth_admin) [A-Z]
# The image's default privileges FOR those roles, and default privileges that grant to them.
A ^defacl\|role (postgres|supabase_auth_admin) schema
A ^defacl\|role [a-z_]+ schema [a-z_]+ type [a-zA-Z]\|[a-z_]+ -> (postgres|dashboard_user|supabase_auth_admin) [A-Z]

# Members of extensions (kind ext-member: only their ACLs are compared). The old image's default privileges in schema public gave
# service_role EXECUTE on the pg_trgm functions of the chain database; the new bootstrap has no default privileges and PUBLIC
# keeps EXECUTE, so nothing is lost (design 13, H2).
A ^acl\|ext-member\|function public\.[a-z_]+\(.*\)\|supabase_admin -> service_role EXECUTE$
# ... and the other side of the same change: on a database without default privileges the pg_trgm functions keep the
# built-in PUBLIC EXECUTE (the old image's default ACL replaced it by an explicit list).
B ^acl\|ext-member\|function public\.(gin_[a-z_]+|gtrgm_[a-z_]+|set_limit|show_limit|show_trgm|(strict_)?word_similarity[a-z_]*|similarity[a-z_]*)\(.*\)\|supabase_admin -> PUBLIC EXECUTE$

# Role attributes are the bootstrap's job, not the baseline's: migration 104 makes supabase_auth_admin NOLOGIN on the
# old image; the new bootstrap has no such role at all.
A ^role\|supabase_auth_admin\|
B@supabase ^role\|supabase_auth_admin\|

# Objects of the OLD IMAGE that the chain removes (migrations 103, 105, 106: pg_graphql, supabase_vault, pg_net, pg_cron,
# Realtime). The baseline never creates them, so the baseline database, which starts from the same old image, still holds
# them; the new image (postgres:17 plus bootstrap) never has them. Old image only: on postgres:17 a line like these means the
# baseline creates something it must not (design 4.7), and it is reported.
# The extensions themselves, their owner, and their comments:
B@supabase ^extension\|(pg_cron|pg_graphql|pg_net|supabase_vault)\|
B@supabase ^comment\|extension (pg_cron|pg_graphql|pg_net|supabase_vault)\|
# The image's helper functions that react to those extensions, and the graphql schemas' functions:
B@supabase ^function\|(extensions\.(grant_pg_cron_access|grant_pg_graphql_access|grant_pg_net_access|set_graphql_placeholder)|graphql(_public)?\.[a-z_]+)\(
B@supabase ^acl\|function\|(extensions\.(grant_pg_cron_access|grant_pg_graphql_access|grant_pg_net_access|set_graphql_placeholder)|graphql(_public)?\.[a-z_]+)\(
B@supabase ^comment\|function extensions\.(grant_pg_cron_access|grant_pg_graphql_access|grant_pg_net_access|set_graphql_placeholder)\(\)\|
# The event triggers of the image's setup:
B@supabase ^eventtrigger\|(graphql_watch_ddl|graphql_watch_drop|issue_graphql_placeholder|issue_pg_cron_access|issue_pg_graphql_access|issue_pg_net_access)\|
# The schemas those extensions and Realtime own, and the default privileges the image gives in them:
B@supabase ^schema\|(graphql|graphql_public|realtime|vault)\|
B@supabase ^acl\|schema\|(graphql|graphql_public|realtime|vault)\|
B@supabase ^defacl\|role [a-z_]+ schema (graphql|graphql_public|realtime) type
# The ACLs of the members of those extensions (vault.secrets, graphql sequences, ...):
B@supabase ^acl\|ext-member\|(function|table|view|sequence) (vault|graphql|graphql_public|extensions)\.
# The Realtime publication of the image:
B@supabase ^publication(-table)?\|supabase_realtime\|

# Expected when B is built on postgres:17 (design 14d).
# PG17 adds the MAINTAIN privilege to table-like ACLs (relations, extension views, default privileges); only for the
# named roles, never for PUBLIC.
B@pg17 ^acl\|(relation|ext-member)\|[^|]+\|[a-z_]+ -> (anon|authenticated|service_role|supabase_admin|supabase_storage_admin) MAINTAIN( WITH GRANT OPTION)?$
B@pg17 ^defacl\|role [a-z_]+ schema [a-z_]+ type r\|[a-z_]+ -> (anon|authenticated|service_role|supabase_admin) MAINTAIN$
# (The old Storage image's grants on storage.migrations to the API roles need no entry: Storage 1.80.2 runs on both sides and
# leaves the same ACL, design 14d expected a difference that the first trial did not find.)

# --- the rest of the old-image-versus-postgres:17 differences found by the first trial (item18-trial.md section 15) -------
# Each is a property of the image or of the PostgreSQL version, not of the baseline.
# The database postgres: created by the old image's initdb as owner postgres; by the bootstrap's initdb as supabase_admin.
A@pg17 ^database\|postgres\|owner postgres 
B@pg17 ^database\|postgres\|owner supabase_admin 
A@pg17 ^acl\|database\|postgres\|
B@pg17 ^acl\|database\|postgres\|
# Schemas extensions and storage: owner and grants come from the old image (postgres, supabase_admin) or from the bootstrap
# (supabase_admin, supabase_storage_admin). Both grant USAGE to the three API roles.
A@pg17 ^schema\|(extensions|storage)\|owner
B@pg17 ^schema\|(extensions|storage)\|owner
A@pg17 ^acl\|schema\|(extensions|storage)\|
B@pg17 ^acl\|schema\|(extensions|storage)\|
# Storage 1.80.2 on the old image kept explicit EXECUTE grants for the API roles on three helper functions (PUBLIC keeps
# EXECUTE on both sides, so the effective access is the same).
A@pg17 ^acl\|function\|storage\.(extension|filename|foldername)\(name text\)\|supabase_storage_admin -> (anon|authenticated|service_role) EXECUTE$
# GoTrue's tables in the old image (the new image has no auth tables; auth.users is only a trial stand-in, see section 15).
A@pg17 ^(column|constraint|index|relation|sequence|seqstate|seed)\|auth\.
A@pg17 ^comment\|(table|sequence) auth\.
# PostgREST's DDL watch of the old image (event triggers and their functions in extensions); the baseline strips the triggers.
A@pg17 ^eventtrigger\|pgrst_(ddl|drop)_watch\|
A@pg17 ^function\|extensions\.pgrst_(ddl|drop)_watch\(\)\|
A@pg17 ^acl\|function\|extensions\.pgrst_(ddl|drop)_watch\(\)\|
# pg_stat_statements: version 1.10 (PG15) versus 1.11 (PG17) and the changed signature of its reset function.
A@pg17 ^extension\|pg_stat_statements\|version 1\.10 
B@pg17 ^extension\|pg_stat_statements\|version 1\.11 
A@pg17 ^acl\|ext-member\|function extensions\.pg_stat_statements_reset\(oid,oid,bigint\)\|
B@pg17 ^acl\|ext-member\|function extensions\.pg_stat_statements_reset\(oid,oid,bigint,boolean\)\|
# The bootstrap's pgbouncer.get_auth never returns a superuser's verifier and pins its search_path (design 14b).
A@pg17 ^function\|pgbouncer\.get_auth\(
B@pg17 ^function\|pgbouncer\.get_auth\(
# PG17 prints the columns of single-table views without the table qualifier (same view, different deparse).
A@pg17 ^viewdef\|public\.(customers_with_priority_contact|high_cpu_analysis|low_stock_alerts|rls_policies_summary|sensor_status)\|
B@pg17 ^viewdef\|public\.(customers_with_priority_contact|high_cpu_analysis|low_stock_alerts|rls_policies_summary|sensor_status)\|
# Roles of the old image that the bootstrap does not create, their memberships, and their settings.
A@pg17 ^role\|(postgres|dashboard_user|supabase_etl_admin|supabase_read_only_user|supabase_replication_admin)\|
A@pg17 ^member\|(postgres|supabase_etl_admin|supabase_read_only_user|dashboard_user|supabase_replication_admin) in 
A@pg17 ^setting\|role (postgres|supabase_auth_admin|supabase_read_only_user) db 
# Role differences by design: pg-safeupdate is not loaded (design 12.3); supabase_storage_admin has INHERIT and no CREATEROLE
# (design 14b); the old image's search_path setting of supabase_admin has a stray backslash in "\$user"; the grantor of
# authenticator's three memberships is postgres in the old image and supabase_admin in the bootstrap.
A@pg17 ^role\|authenticator\|.*session_preload_libraries=safeupdate
B@pg17 ^role\|authenticator\|.*"rolconfig": \["statement_timeout=8s", "lock_timeout=8s"\]
A@pg17 ^setting\|role authenticator db \*\|session_preload_libraries=safeupdate
A@pg17 ^role\|supabase_storage_admin\|.*"rolinherit": false.*"rolcreaterole": true
B@pg17 ^role\|supabase_storage_admin\|.*"rolinherit": true.*"rolcreaterole": false
A@pg17 ^role\|supabase_admin\|.*search_path=\\"\\\\\$user
B@pg17 ^role\|supabase_admin\|.*"search_path=\\"\$user\\", public, auth, extensions"
A@pg17 ^setting\|role supabase_admin db \*\|search_path="\\\$user"
B@pg17 ^setting\|role supabase_admin db \*\|search_path="\$user"
A@pg17 ^member\|authenticator in (anon|authenticated|service_role)\|admin f grantor postgres$
B@pg17 ^member\|authenticator in (anon|authenticated|service_role)\|admin f grantor supabase_admin$
